function (initial, createFrameReceiver, createComponentLayoutTools, installComponentLayoutState) {
  "use strict";
  const root = document.documentElement;
  const scope = { channel: initial.channel, boardId: initial.boardId, sessionId: initial.sessionId };
  let snapshot = initial;
  let state = initial.userState;
  let localStateDirty = false;
  let selectionMode = "off";
  let deferredSelection = null;
  let queued = null;
  let applying = false;
  let captureBusy = false;
  let selectionSession = null;
  let suppressPointerClick = false;
  let dismissedPointer = null;
  let lastAnchor;
  let pointer = null;
  let textPointer = null;
  let overlay = null;
  let lastGesture = -Infinity;
  let lastGesturePosition;
  let patchFailure = null;
  let pendingMarkerFocus = null;
  let sequence = 0;
  const applied = new Set();
  const dataListeners = new Set();
  const dataWatches = new Map();
  const stateListeners = new Set();
  const pendingAsks = new Map();
  const revealListeners = new Set();
  let markerReveal = null;
  const post = (message) => window.parent.postMessage({ ...message, ...scope }, "*");
  const copy = (value) => value === undefined ? undefined : structuredClone(value);
  const error = (code, message, patchId) => post({ type: "error", error: { code, message: String(message).slice(0, 4000), ...(patchId ? { patchId } : {}) } });
  const emitListeners = (listeners, value) => { for (const listener of listeners) { try { listener(copy(value)); } catch (e) { error("listener-error", e.message); } } };
  const currentAnchor = () => ({ kind: "board", boardRevision: snapshot.boardRevision, canvasRevision: snapshot.canvas.revision });
  const acting = () => !markerReveal && performance.now() - lastGesture < 1800 && navigator.userActivation && navigator.userActivation.isActive;
  const cancelAsk = { status: "cancelled", outcome: null };
  const ask = (input) => {
    const request = typeof input === "string" ? { prompt: input } : input;
    if (!request || typeof request.prompt !== "string" || !request.prompt.trim() || request.prompt.length > 24000) return Promise.reject(new TypeError("board.ask requires a prompt of 1–24000 characters"));
    if (!acting() || pendingAsks.size >= 8 || selectionMode !== "off") return Promise.resolve(cancelAsk);
    lastGesture = -Infinity;
    const id = `ask-${++sequence}`;
    return new Promise((resolve) => {
      const timer = setTimeout(() => { pendingAsks.delete(id); resolve(cancelAsk); }, 120000);
      pendingAsks.set(id, { resolve, timer });
      post({ type: "ask", id, request: { prompt: request.prompt, anchor: bindAuthoredAnchor(request.anchor || lastAnchor || currentAnchor()), kind: request.kind || "question", position: lastGesturePosition } });
    });
  };
  const managedViewNamespace = "__paseoContainerViews";
  const writeState = (value) => {
    if (!value || typeof value !== "object" || Array.isArray(value)) throw new TypeError("board.saveState requires a JSON object");
    const serialized = JSON.stringify(value);
    if (serialized.length > 524288) throw new RangeError("board state exceeds 512 KiB");
    state = JSON.parse(serialized);
    localStateDirty = true;
    post({ type: "state", state });
    return copy(state);
  };
  const saveState = (value) => {
    if (!value || typeof value !== "object" || Array.isArray(value)) throw new TypeError("board.saveState requires a JSON object");
    const next = { ...value };
    // Authored state still uses replacement semantics. Its old/full snapshots
    // cannot overwrite the independently maintained container view records.
    if (Object.prototype.hasOwnProperty.call(state, managedViewNamespace)) next[managedViewNamespace] = state[managedViewNamespace];
    else delete next[managedViewNamespace];
    return writeState(next);
  };
  const subscribe = (listeners, fn) => {
    if (typeof fn !== "function") throw new TypeError("listener must be a function");
    listeners.add(fn);
    return () => listeners.delete(fn);
  };
  // Watch only selected JSON values. Changed paths filter unrelated subscriptions.
  const watchSegments = (path) => {
    if (typeof path !== "string" || path.length > 2000 || (path !== "" && !path.startsWith("/")) || /~(?![01])/.test(path)) throw new TypeError("watch requires a JSON Pointer (empty string means all data)");
    const parts = path === "" ? [] : path.slice(1).split("/").map(part => part.replace(/~1/g, "/").replace(/~0/g, "~"));
    if (parts.length > 64 || parts.some(part => ["__proto__", "prototype", "constructor"].includes(part))) throw new TypeError("watch path is not allowed");
    return parts;
  };
  const readWatchedValue = (data, parts) => {
    let value = data;
    for (const part of parts) {
      if (!value || typeof value !== "object" || !Object.prototype.hasOwnProperty.call(value, part)) return { exists: false };
      if (Array.isArray(value) && !/^(0|[1-9][0-9]*)$/.test(part)) return { exists: false };
      value = value[part];
    }
    return { exists: true, value };
  };
  const equalJson = (a, b) => {
    if (a === b) return true;
    if (!a || !b || typeof a !== "object" || typeof b !== "object" || Array.isArray(a) !== Array.isArray(b)) return false;
    const keys = Object.keys(a);
    return keys.length === Object.keys(b).length && keys.every(key => Object.prototype.hasOwnProperty.call(b, key) && equalJson(a[key], b[key]));
  };
  const deliverWatch = (entry, listener, immediate = false) => {
    if (!listener.active) return;
    if (listener.schedule === "frame" && !immediate) {
      if (listener.frame === null) listener.frame = requestAnimationFrame(() => { listener.frame = null; deliverWatch(entry, listener, true); });
      return;
    }
    try { listener.fn(entry.current.exists ? copy(entry.current.value) : undefined, Object.freeze({ path: entry.path, exists: entry.current.exists, boardRevision: snapshot.boardRevision, canvasRevision: snapshot.canvas.revision })); }
    catch (e) { error("listener-error", e instanceof Error ? e.message : String(e)); }
  };
  const emitWatches = (changedPaths) => {
    const deliveries = [];
    // Update every cache before calling authored code. A callback may subscribe
    // to another existing path; its initial value must belong to this snapshot.
    for (const entry of Array.from(dataWatches.values())) {
      if (changedPaths && !changedPaths.some(path => path === entry.path || path.startsWith(entry.path + "/") || entry.path.startsWith(path + "/"))) continue;
      const next = readWatchedValue(snapshot.canvas.data, entry.parts);
      if (next.exists === entry.current.exists && equalJson(next.value, entry.current.value)) continue;
      entry.current = next;
      deliveries.push({ entry, listeners: Array.from(entry.listeners) });
    }
    for (const { entry, listeners } of deliveries) {
      for (const listener of listeners) if (entry.listeners.has(listener)) deliverWatch(entry, listener);
    }
  };
  const watch = (path, fn, options) => {
    const parts = watchSegments(path);
    if (typeof fn !== "function") throw new TypeError("watch listener must be a function");
    if (options !== undefined && (!options || typeof options !== "object" || options.schedule !== "frame")) throw new TypeError("watch options require schedule: 'frame'");
    let entry = dataWatches.get(path);
    if (!entry) { entry = { path, parts, current: readWatchedValue(snapshot.canvas.data, parts), listeners: new Set() }; dataWatches.set(path, entry); }
    // Frame subscriptions are latest-state rendering, not a lossless event stream.
    const listener = { fn, schedule: options?.schedule, frame: null, active: true };
    entry.listeners.add(listener);
    deliverWatch(entry, listener, true);
    return () => { listener.active = false; if (listener.frame !== null) cancelAnimationFrame(listener.frame); entry.listeners.delete(listener); if (!entry.listeners.size && dataWatches.get(path) === entry) dataWatches.delete(path); };
  };
  // No SDK or execution primitive is exposed. Data and view state remain task-agnostic JSON.
  Object.defineProperty(window, "board", { configurable: false, writable: false, value: Object.freeze({
    getData: () => copy(snapshot.canvas.data),
    onData: (fn) => subscribe(dataListeners, fn),
    watch,
    getState: () => copy(state),
    restoreState: () => copy(state),
    onState: (fn) => subscribe(stateListeners, fn),
    saveState,
    ask,
    onReveal: (fn) => subscribe(revealListeners, fn),
    getContext: () => ({ boardId: scope.boardId, documentVersion: initial.canvas.documentVersion, boardRevision: snapshot.boardRevision, canvasRevision: snapshot.canvas.revision }),
  }) });
  window.openai = Object.freeze({ sendFollowUpMessage: ask });
  const componentLayoutTools = createComponentLayoutTools();
  const componentLayoutState = installComponentLayoutState({
    getState: () => copy(state), saveState: writeState, onState: fn => subscribe(stateListeners, fn),
  }, initial.canvas.documentVersion);

  let appliedTheme = initial.theme;
  const applyTheme = (theme) => {
    if (!theme || !["light", "dark"].includes(theme.scheme) || !theme.tokens || typeof theme.tokens !== "object") return;
    if (equalJson(appliedTheme, theme)) return;
    if (appliedTheme?.scheme !== theme.scheme) root.setAttribute("data-theme", theme.scheme);
    // The host scheme is a CSS-layer default. An authored fixed light/dark canvas may
    // override color-scheme without each host refresh overwriting it with inline style.
    for (const name of Object.keys(appliedTheme?.tokens || {})) if (!Object.prototype.hasOwnProperty.call(theme.tokens, name)) root.style.removeProperty("--" + name);
    for (const [name, value] of Object.entries(theme.tokens)) if (appliedTheme?.tokens[name] !== value && /^host-[a-z0-9-]+$/.test(name) && typeof value === "string" && /^[#a-zA-Z0-9(),.%+*/ -]{1,200}$/.test(value)) root.style.setProperty("--" + name, value);
    appliedTheme = theme;
    window.dispatchEvent(new CustomEvent("paseo:themechange", { detail: { theme: theme.scheme, palette: "custom" } }));
  };

  const runScript = (old) => new Promise((resolve, reject) => {
    const script = document.createElement("script");
    for (const attribute of old.attributes) script.setAttribute(attribute.name, attribute.value);
    script.textContent = old.textContent;
    const executable = !script.type || /^(module|(?:text|application)\/(?:java|ecma)script)$/i.test(script.type);
    if (!executable) { old.replaceWith(script); resolve(); return; }
    let scriptError;
    let timer;
    const finish = (failure) => { clearTimeout(timer); window.removeEventListener("error", onError); failure ? reject(failure) : resolve(); };
    const onError = (event) => { scriptError = new Error(event.message || "Patch script failed"); };
    window.addEventListener("error", onError);
    const external = script.src || script.type === "module";
    if (external) {
      script.async = false;
      script.addEventListener("load", () => finish(scriptError), { once: true });
      script.addEventListener("error", () => finish(new Error("Patch script could not load")), { once: true });
      timer = setTimeout(() => finish(new Error("Patch script timed out; it will not be retried")), 20000);
    }
    try { old.replaceWith(script); if (!external) finish(scriptError); } catch (e) { finish(e); }
  });
  const componentTransferFailure = (operation, node, destination = null, position = "append") => ["move", "copy"].includes(operation)
    ? componentLayoutTools.transferFailure(operation, node, destination, position) : null;
  const applyPatch = async (patch) => {
    if (applied.has(patch.id)) return;
    // Claim before mutation. A partial script failure must never run the same patch a second time.
    applied.add(patch.id);
    const targets = document.querySelectorAll(patch.selector);
    if (targets.length !== 1) throw new Error(`Patch selector must match exactly one element; matched ${targets.length}`);
    const target = targets[0];
    if (target === root || target === document.head || target === document.body) {
      if (["replace", "remove", "move", "copy", "text", "layout"].includes(patch.operation)) throw new Error("Replacing document roots requires publishing a new document version");
    }
    if (["move", "copy", "text", "layout"].includes(patch.operation)) {
      if (!matchesComponentIdentity(target, patch.sourceIdentity)) throw new Error("The edited component structure changed; reload its current authored version");
      const transferFailure = componentTransferFailure(patch.operation, target);
      if (transferFailure) throw new Error(transferFailure);
      if (patch.operation === "layout") {
        const focused = document.activeElement;
        const restoreFocus = focused instanceof HTMLElement && target.contains(focused);
        const caret = restoreFocus && typeof focused.selectionStart === "number" ? [focused.selectionStart, focused.selectionEnd, focused.selectionDirection] : null;
        capturePatchTargets(patch, target);
        componentLayoutTools.apply(target, patch.layout, patch.id, { move(parent, node, before) {
          if (typeof parent.moveBefore === "function" && node.isConnected && parent.isConnected) parent.moveBefore(node, before);
          else parent.insertBefore(node, before);
        } });
        componentLayoutState.refresh();
        if (restoreFocus && focused.isConnected && document.activeElement !== focused) focused.focus({ preventScroll: true });
        if (caret) try { focused.setSelectionRange(...caret); } catch {}
        return;
      }
      if (patch.operation === "text") {
        const textNode = patch.textNodeIndex === undefined ? null : target.childNodes[patch.textNodeIndex];
        if (textNode ? textNode.nodeType !== Node.TEXT_NODE || textNode.textContent !== patch.expectedText : target.children.length || target.textContent !== patch.expectedText) throw new Error("The edited text changed before this patch arrived");
        capturePatchTargets(patch, target);
        if (textNode) textNode.textContent = patch.text; else target.textContent = patch.text;
        return;
      }
      const destinations = document.querySelectorAll(patch.destination);
      if (destinations.length !== 1) throw new Error("Edit destination must match exactly one element");
      const destination = destinations[0];
      if (!matchesComponentIdentity(destination, patch.destinationIdentity) || target === destination || target.contains(destination)) throw new Error("The edit destination changed or contains a cycle");
      const placementFailure = componentTransferFailure(patch.operation, target, destination, patch.position);
      if (placementFailure) throw new Error(placementFailure);
      const parent = ["before", "after"].includes(patch.position) ? destination.parentNode : destination;
      const before = patch.position === "before" ? destination : patch.position === "after" ? destination.nextSibling : patch.position === "prepend" ? destination.firstChild : null;
      if (patch.operation === "copy") {
        // Parse in the source's actual namespace/context. An HTML template turns SVG
        // subcomponents such as <g>/<rect> into invisible HTML unknown elements. Range
        // parsing also retains SVG foreignObject's HTML integration points.
        const range = document.createRange(); range.selectNode(target);
        const content = range.createContextualFragment(patch.html);
        // Copy authoritative authored markup, never runtime listeners/closures. Defense in
        // depth: copied scripts and inline handler attributes never execute on insertion.
        for (const script of content.querySelectorAll("script")) script.remove();
        for (const element of content.querySelectorAll("*")) for (const attribute of Array.from(element.attributes)) if (/^on/i.test(attribute.name)) element.removeAttribute(attribute.name);
        const inserted = Array.from(content.children); capturePatchTargets(patch, target, inserted);
        parent.insertBefore(content, before); return;
      }
      const focused = document.activeElement;
      const restoreFocus = focused instanceof HTMLElement && target.contains(focused);
      const caret = restoreFocus && typeof focused.selectionStart === "number" ? [focused.selectionStart, focused.selectionEnd, focused.selectionDirection] : null;
      capturePatchTargets(patch, target);
      // New browsers preserve iframe/media/focus state with a same-document atomic move.
      // The fallback still retains the actual node and listeners, then restores input caret.
      if (typeof parent.moveBefore === "function" && target.isConnected && parent.isConnected) parent.moveBefore(target, before);
      else parent.insertBefore(target, before);
      if (restoreFocus && focused.isConnected && document.activeElement !== focused) focused.focus({ preventScroll: true });
      if (caret) try { focused.setSelectionRange(...caret); } catch {}
      return;
    }
    if (patch.operation === "remove") { capturePatchTargets(patch, target); target.remove(); return; }
    const template = document.createElement("template");
    template.innerHTML = patch.html;
    const scripts = Array.from(template.content.querySelectorAll("script"));
    capturePatchTargets(patch, target, Array.from(template.content.children));
    if (patch.operation === "append") target.append(template.content);
    else if (patch.operation === "prepend") target.prepend(template.content);
    else if (patch.operation === "replace") target.replaceWith(template.content);
    else throw new Error("Unknown patch operation");
    for (const script of scripts) if (script.isConnected) await runScript(script);
  };
  const receiver = createFrameReceiver(initial, request => post({ type: "snapshot-request", ...request }), componentLayoutTools.validate);
  let versionedTransport = false;
  const enqueue = (update) => {
    if (!update) return;
    queued = queued ? { snapshot: update.snapshot,
      changedPaths: queued.changedPaths === null || update.changedPaths === null ? null : [...new Set([...queued.changedPaths, ...update.changedPaths])],
      patches: [...new Map([...queued.patches, ...update.patches].map(patch => [patch.id, patch])).values()] } : update;
    if (componentLayoutPreview && update.snapshot.canvas.revision !== componentLayoutPreview.revision) { clearComponentLayoutPreview(); setComponentNotice("画布已更新，布局预览已撤下；请取消后重新选择"); }
    void drain();
  };
  const receiveSnapshot = (next, version) => {
    if (version !== undefined) { versionedTransport = true; enqueue(receiver.snapshot(next, version)); return; }
    // Compatibility with older hosts. Versioned traffic exclusively uses the validated receiver.
    if (versionedTransport) return;
    if (!next || !next.canvas || next.canvas.documentVersion !== initial.canvas.documentVersion || next.boardRevision < (queued?.snapshot || snapshot).boardRevision) return;
    enqueue({ snapshot: next, changedPaths: null, patches: next.canvas.patches });
  };
  const drain = async () => {
    if (applying || selectionMode !== "off" || selectionSession || captureBusy) return;
    if (!queued) { reportCanvasActivity(); drainMarkerFocus(); return; }
    applying = true;
    reportCanvasActivity();
    try {
      while (queued && selectionMode === "off" && !selectionSession && !captureBusy) {
        const update = queued;
        const next = update.snapshot;
        queued = null;
        const dataChanged = update.changedPaths === null ? !equalJson(snapshot.canvas.data, next.canvas.data) : update.changedPaths.length > 0;
        const serverStateChanged = !equalJson(snapshot.userState, next.userState);
        snapshot = next;
        applyTheme(next.theme);
        if (dataChanged) {
          emitWatches(update.changedPaths);
          emitListeners(dataListeners, next.canvas.data);
          window.dispatchEvent(new CustomEvent("board:data", { detail: copy(next.canvas.data) }));
        }
        if (localStateDirty) {
          // A slow acknowledgement for an earlier save must never overwrite newer local input.
          if (equalJson(next.userState, state)) localStateDirty = false;
        } else if (serverStateChanged && !equalJson(next.userState, state)) {
          state = next.userState;
          emitListeners(stateListeners, state);
        }
        if (patchFailure) continue;
        for (const patch of update.patches) {
          try { await applyPatch(patch); }
          catch (e) {
            patchFailure = patch.id;
            error("patch-failed", `${e.message}. Later patches are paused until the document is republished.`, patch.id);
            break;
          }
        }
      }
    } finally {
      applying = false;
      reportCanvasActivity();
      queueMarkerPositions();
      if (deferredSelection !== null) { const mode = deferredSelection; deferredSelection = null; setSelectionMode(mode); }
      drainMarkerFocus();
      resumeComponentEdit();
    }
  };

  // Selection is opt-in. In ordinary mode we never consume pointer, keyboard or drag events.
  const selectorFor = (node) => {
    if (!(node instanceof Element)) return undefined;
    if (node.id) return "#" + CSS.escape(node.id);
    const parts = [];
    while (node && node !== document.body && parts.length < 12) {
      if (node.id) { parts.unshift("#" + CSS.escape(node.id)); break; }
      const name = node.localName;
      if (!name) break;
      const siblings = node.parentElement ? Array.from(node.parentElement.children).filter((n) => n.localName === name) : [];
      parts.unshift(name + (siblings.length > 1 ? `:nth-of-type(${siblings.indexOf(node) + 1})` : ""));
      node = node.parentElement;
    }
    return parts.length ? parts.join(" > ").slice(0, 1000) : undefined;
  };
  const viewport = () => ({ width: innerWidth, height: innerHeight, scrollX, scrollY });
  const rect = (r) => ({ x: r.left + scrollX, y: r.top + scrollY, width: r.width, height: r.height });
  // Identity metadata is optional and never changes the author's DOM or the original
  // quote/PNG/coordinates. A live reference survives reparenting; stable IDs survive patches.
  const bindingNodes = new Map();
  const bindingFingerprint = (node) => {
    const text = (node.textContent || "").replace(/\s+/g, " ").trim().slice(0, 1000);
    const input = [node.localName, node.getAttribute("role"), node.getAttribute("name"), node.getAttribute("type"), node.getAttribute("aria-label"), text].join("\n");
    let hash = 2166136261;
    for (let i = 0; i < input.length; i++) hash = Math.imul(hash ^ input.charCodeAt(i), 16777619);
    return (hash >>> 0).toString(16).padStart(8, "0");
  };
  const authorElement = (node) => node instanceof Element && node !== root && node !== document.body && node !== document.head && !node.closest("[data-board-overlay]");
  const identityFor = (node) => {
    const selector = selectorFor(node);
    if (!selector || !authorElement(node)) return null;
    return { selector, tag: node.localName, ...(node.id && node.id.length <= 1000 ? { id: node.id } : {}), fingerprint: bindingFingerprint(node) };
  };
  const captureBinding = (target, region, enclose = false) => {
    if (!(target instanceof Element)) return undefined;
    // A region may span several components; select its enclosing context, not the element
    // where the drag happened to start. Whole-page regions remain geometric references.
    if (enclose) {
      while (authorElement(target)) {
        const r = rect(target.getBoundingClientRect());
        if (region.x >= r.x - 1 && region.y >= r.y - 1 && region.x + region.width <= r.x + r.width + 1 && region.y + region.height <= r.y + r.height + 1) break;
        target = target.parentElement;
      }
    }
    const identity = identityFor(target);
    if (!identity) return undefined;
    const r = rect(target.getBoundingClientRect());
    const ancestors = [];
    const nodes = [target];
    for (let parent = target.parentElement; authorElement(parent) && ancestors.length < 8; parent = parent.parentElement) {
      const ancestor = identityFor(parent);
      if (ancestor) { ancestors.push(ancestor); nodes.push(parent); }
    }
    const key = `binding-${++sequence}-${scope.sessionId}`.slice(0, 160);
    const binding = { documentVersion: snapshot.canvas.documentVersion, key, target: identity, ancestors,
      relative: { x: Math.max(0, Math.min(1, r.width ? (region.x - r.x) / r.width : 0)), y: Math.max(0, Math.min(1, r.height ? (region.y - r.y) / r.height : 0)) } };
    bindingNodes.set(key, nodes);
    // Unsent selections need no long-lived registry. Active marker bindings are retained.
    if (bindingNodes.size > 600) {
      const retained = new Set(markerItems.map(item => item.marker.anchor?.binding?.key));
      for (const candidate of bindingNodes.keys()) if (candidate !== key && !retained.has(candidate)) { bindingNodes.delete(candidate); if (bindingNodes.size <= 500) break; }
    }
    return binding;
  };
  const bindAuthoredAnchor = (anchor) => {
    if (!anchor || anchor.binding || !anchor.rect || !anchor.selector || anchor.canvasRevision !== snapshot.canvas.revision || anchor.boardRevision !== snapshot.boardRevision) return anchor;
    try {
      const targets = document.querySelectorAll(anchor.selector);
      if (targets.length === 1) {
        const binding = captureBinding(targets[0], anchor.rect, anchor.kind === "region");
        if (binding) return { ...anchor, binding };
      }
    } catch {}
    return anchor;
  };
  const resolveIdentity = (identity, cached, allowPath) => {
    if (!identity) return null;
    if (identity.id) {
      let matches;
      try { matches = document.querySelectorAll("#" + CSS.escape(identity.id)); } catch { return null; }
      // Duplicate IDs are ambiguous even when one original node is still in memory.
      if (matches.length !== 1 || matches[0].localName !== identity.tag) return null;
      return matches[0];
    }
    if (cached) return cached.isConnected && authorElement(cached) ? cached : null;
    if (!allowPath) return null;
    try {
      const matches = document.querySelectorAll(identity.selector);
      if (matches.length === 1 && authorElement(matches[0]) && matches[0].localName === identity.tag && bindingFingerprint(matches[0]) === identity.fingerprint) return matches[0];
    } catch {}
    return null;
  };
  const visibleBox = (node, revealScroll = false) => {
    if (!authorElement(node) || !node.isConnected || !node.getClientRects().length) return null;
    const style = getComputedStyle(node);
    if (style.visibility !== "visible" || style.display === "none" || style.contentVisibility === "hidden") return null;
    const box = node.getBoundingClientRect();
    let left = box.left, top = box.top, right = box.right, bottom = box.bottom;
    // CSS accordions and nested scrollers can keep a child's layout box while completely
    // clipping its pixels. Only visible pixels can host a marker; viewport clipping itself
    // is excluded here because a sidebar action is allowed to scroll an offscreen component.
    for (let parent = node.parentElement; authorElement(parent); parent = parent.parentElement) {
      const css = getComputedStyle(parent);
      if (css.contentVisibility === "hidden") return null;
      if (parent.localName === "details" && !parent.open) {
        const summary = Array.from(parent.children).find(child => child.localName === "summary");
        if (!summary?.contains(node)) return null;
      }
      const clipX = /^(hidden|clip|auto|scroll)$/.test(css.overflowX) && !(revealScroll && /^(auto|scroll)$/.test(css.overflowX));
      const clipY = /^(hidden|clip|auto|scroll)$/.test(css.overflowY) && !(revealScroll && /^(auto|scroll)$/.test(css.overflowY));
      if (!clipX && !clipY) continue;
      const r = parent.getBoundingClientRect();
      const scaleX = parent.offsetWidth ? r.width / parent.offsetWidth : 1;
      const scaleY = parent.offsetHeight ? r.height / parent.offsetHeight : 1;
      if (clipX) { const x = r.left + parent.clientLeft * scaleX; left = Math.max(left, x); right = Math.min(right, x + parent.clientWidth * scaleX); }

    for (let node = hiddenContent ? hiddenContent.parentElement : target; authorElement(node); node = node.parentElement) if (visibleBox(node)) return node;

    return tab;

    // ancestors never fall back to a visible dashboard shell. Native closed details
    // and scroll/accordion clipping remain distinguishable and may use their summary.
    if (getComputedStyle(target).visibility !== "visible") return target;

      if (node.getAttribute("role") === "tabpanel" && tabForPanel(node)?.getAttribute("aria-selected") === "false") return node;
    }

    bindingNodes.set(binding.key, nodes.map((node, index) => node || cached[index]));

      }
      return { rect: anchor.rect, mode: "snapshot", tracked: nodes.filter(Boolean) };

    return { rect: anchor?.rect, mode: "snapshot", tracked: [] };

    }
    if (!isEditor(event) && event.shiftKey && (event.metaKey || event.ctrlKey) && !event.altKey) { componentShortcut = true; updateComponentHover(); }
    if (inComponents(event) && event.key !== "Escape") { event.stopImmediatePropagation(); return; }
    if (inMarkers(event)) { event.stopImmediatePropagation(); return; }
    if (event.target instanceof Element) { const r = event.target.getBoundingClientRect(); lastGesturePosition = { x: r.left, y: r.bottom }; }
    if (event.key === "Escape" && (selectionSession || selectionMode !== "off" || lastAnchor)) {
      event.preventDefault(); event.stopImmediatePropagation();
      if (componentPending) return;
      componentClipboard = null; componentNotice = ""; componentShortcut = false;
      selectionMode = "off"; cancelSelection({ notify: true }); post({ type: "selection-end" }); return;
    }
  }, true);
  for (const type of ["beforeinput","input","compositionstart","compositionupdate","compositionend","paste","copy","cut","drop","dragstart","pointerup","dblclick","mousedown","mouseup","keypress","focus","blur","focusin","focusout"]) window.addEventListener(type,event=>{
    if (!inComponentText(event)) return;
    if (type === "compositionstart") componentText.composing=true;
    if (type === "compositionend") {componentText.composing=false;syncComponentText();}
    if (type === "input") syncComponentText();
    if (type === "beforeinput") {
      if (componentPending || componentText.invalid || /^format|insertHTML|insertFromDrop/.test(event.inputType)) event.preventDefault();
      else if (event.isTrusted && /^(insert|delete|history)/.test(event.inputType)) {
        // Preserve actual spaces once editing begins. With collapsed whitespace,
        // Chromium rewrites ordinary boundary spaces to NBSP in the DOM. Changing
        // the active run before the edit keeps the original input bytes and native
        // undo; an untouched run still uses the source's exact wrapping above.
        const lineBreak=["insertParagraph","insertLineBreak"].includes(event.inputType) || /[\r\n]/.test(event.data || "");
        // Let the browser represent new lines as blocks/BRs rather than its
        // pre-wrap-only terminal newline sentinel. After input, the draft returns
        // to pre-wrap; its semantic reader preserves every real block boundary.
        componentText.run.editor.style.setProperty("white-space",lineBreak?"normal":"pre-wrap","important");
        componentText.run.editor.style.setProperty("white-space-collapse",lineBreak?"collapse":"preserve","important");
      }
    }
    if (type === "paste") {
      event.preventDefault();
      if (!componentPending && !componentText.invalid) {
        const plain=event.clipboardData?.getData("text/plain")||"";
        if (plain.length>24000) setComponentNotice("这段文字太长，请缩短后重试");
        else document.execCommand("insertText",false,plain);
      }
    }
    // Only plain text enters this isolated draft. Drops are disabled rather than
    // allowing the browser to move authored nodes or import clipboard HTML.
    if (type === "drop" || type === "dragstart") event.preventDefault();
    event.stopImmediatePropagation();
  },true);
  window.addEventListener("keyup", (event) => {
    if (inComponentText(event)) { event.stopImmediatePropagation(); return; }
    if (!event.isTrusted) return;
    componentShortcut = event.shiftKey && (event.metaKey || event.ctrlKey) && !event.altKey && !isEditor(event);
    if (componentDrag) return;
    if (!componentShortcut) clearComponentHover(); else updateComponentHover();
  }, true);
  window.addEventListener("message", (event) => {
    const message = event.data;
    if (!event.isTrusted || event.source !== window.parent || !message || message.channel !== scope.channel || message.boardId !== scope.boardId || message.sessionId !== scope.sessionId) return;
    if (message.type === "snapshot") receiveSnapshot(message.snapshot, message.version);
    else if (message.type === "delta") { versionedTransport = true; enqueue(receiver.delta(message.delta)); }
    else if (message.type === "interactions" && Array.isArray(message.markers)) renderMarkers(message.markers, message.selectedId);
    else if (message.type === "focus-update") focusUpdate(message);
    else if (message.type === "host-activity" && typeof message.composing === "boolean") { hostComposing = message.composing; reportCanvasActivity(); }
    else if (message.type === "focus-interaction" && typeof message.id === "string") focusInteraction(message.id);
    else if (message.type === "selection") setSelectionMode(message.mode);
    else if (message.type === "cancel-capture" && message.selectionId === selectionSession?.id) cancelSelection({ clear: false });
    else if (message.type === "clear-selection") { selectionMode = "off"; cancelSelection(); }
    else if (message.type === "component-edit-result") receiveComponentEdit(message);
    else if (message.type === "ask-result") {
      const pending = pendingAsks.get(message.id);
      if (pending) { pendingAsks.delete(message.id); clearTimeout(pending.timer); pending.resolve(message.result && message.result.status === "submitted" ? message.result : cancelAsk); }
    }
  });
  const reportedErrors = new Set();
  window.addEventListener("error", (event) => {
    const message = event.message || "Canvas resource could not load";
    if (reportedErrors.has(message) || reportedErrors.size >= 12) return;
    reportedErrors.add(message);
    error("runtime-error", message);
  });
  window.addEventListener("unhandledrejection", (event) => {
    const message = event.reason && event.reason.message || String(event.reason || "Unhandled promise rejection");
    if (reportedErrors.has(message) || reportedErrors.size >= 12) return;
    reportedErrors.add(message);
    error("runtime-error", message);
  });
  const ready = () => {
    post({ type: "ready", capabilities: { frameDelta: 1 } });
    receiveSnapshot(initial);
  };
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", ready, { once: true });
  else ready();
}
